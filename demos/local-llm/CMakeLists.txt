# Dolly's compiler consumes source directly; it does not expose Clang's PCH driver.
function(without_pch_flags directory)
  get_property(options DIRECTORY "${directory}" PROPERTY COMPILE_OPTIONS)
  list(FILTER options EXCLUDE REGEX "fno-pch-timestamp")
  set_property(DIRECTORY "${directory}" PROPERTY COMPILE_OPTIONS "${options}")
  get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(target IN LISTS targets)
    get_property(options TARGET "${target}" PROPERTY COMPILE_OPTIONS)
    list(FILTER options EXCLUDE REGEX "fno-pch-timestamp")
    set_property(TARGET "${target}" PROPERTY COMPILE_OPTIONS "${options}")
  endforeach()
  get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
  foreach(child IN LISTS children)
    without_pch_flags("${child}")
  endforeach()
endfunction()
without_pch_flags("${CMAKE_CURRENT_SOURCE_DIR}/upstream")

# Embed the unchanged upstream shader sources at configure time in Dolly.
set(shader_dir "${CMAKE_CURRENT_SOURCE_DIR}/upstream/ggml/src/ggml-webgpu/wgsl-shaders")
function(expand_shader path output)
  file(READ "${path}" code)
  string(REGEX MATCHALL "#include \"[^\"]+\"" includes "${code}")
  foreach(include IN LISTS includes)
    string(REGEX REPLACE "#include \"([^\"]+)\"" "\\1" name "${include}")
    expand_shader("${shader_dir}/${name}" included)
    string(REPLACE "${include}" "${included}" code "${code}")
  endforeach()
  set(${output} "${code}" PARENT_SCOPE)
endfunction()
file(GLOB shaders "${shader_dir}/*.wgsl")
set(header "${CMAKE_CURRENT_BINARY_DIR}/ggml-wgsl-shaders.hpp")
file(WRITE "${header}" "#pragma once\n#include <string>\n")
foreach(shader IN LISTS shaders)
  get_filename_component(name "${shader}" NAME_WE)
  expand_shader("${shader}" code)
  file(APPEND "${header}" "const char *wgsl_${name} = R\"wgsl(${code})wgsl\";\n")
endforeach()
add_library(ggml-webgpu STATIC upstream/ggml/src/ggml-webgpu/ggml-webgpu.cpp)
