Minimal custom recipe

DOLLY 1
IMAGE my-tools
EXTENDS default

RUN echo 'int main(void) { return 0; }' > /tmp/tool.c
RUN cc /tmp/tool.c -o /usr/bin/tool
CHECK tool
KEEP /usr/bin/tool
ENTRY /usr/bin/pi --no-session
